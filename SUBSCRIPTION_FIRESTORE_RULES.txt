// ============================================
// FIRESTORE SECURITY RULES - SUBSCRIPTION SYSTEM
// Copy và paste vào Firebase Console > Firestore > Rules
// ============================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper function: Kiểm tra user có Premium không
    function hasPremiumAccess(userId) {
      let userData = get(/databases/$(database)/documents/users/$(userId)).data;
      let now = request.time;
      
      // Kiểm tra trial còn hiệu lực
      let trialActive = userData.trialExpiryDate != null && 
                        now < userData.trialExpiryDate;
      
      // Kiểm tra premium còn hiệu lực
      let premiumActive = userData.isPremium == true && 
                          userData.premiumExpiryDate != null && 
                          now < userData.premiumExpiryDate;
      
      return trialActive || premiumActive;
    }
    
    // ============================================
    // USERS COLLECTION - Subscription Data
    // ============================================
    match /users/{userId} {
      // Cho phép user đọc thông tin subscription của chính mình
      allow read: if request.auth != null && request.auth.uid == userId;
      
      // Cho phép tạo user mới với trial
      allow create: if request.auth != null && 
                       request.auth.uid == userId &&
                       request.resource.data.keys().hasAll([
                         'createdAt', 
                         'trialExpiryDate', 
                         'isPremium'
                       ]) &&
                       request.resource.data.isPremium == false;
      
      // Chỉ cho phép update một số trường nhất định
      // (premiumExpiryDate và isPremium chỉ được update bởi admin/cloud function)
      allow update: if request.auth != null && 
                       request.auth.uid == userId &&
                       // Không cho phép user tự update isPremium hoặc premiumExpiryDate
                       (!request.resource.data.diff(resource.data).affectedKeys()
                         .hasAny(['isPremium', 'premiumExpiryDate', 'trialExpiryDate']));
    }
    
    // ============================================
    // PROFILES COLLECTION - Public Access
    // ============================================
    match /profiles/{profileId} {
      // Profiles luôn public (không cần Premium)
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == profileId;
    }
    
    // ============================================
    // POSTS COLLECTION - Public Access
    // ============================================
    match /posts/{postId} {
      // Posts luôn public (không cần Premium)
      allow read: if request.auth != null;
      allow create: if request.auth != null && 
                       request.resource.data.authorUid == request.auth.uid;
      allow update, delete: if request.auth != null && 
                               resource.data.authorUid == request.auth.uid;
    }
    
    // ============================================
    // DOCUMENTS COLLECTION - Public Access
    // ============================================
    match /documents/{documentId} {
      // Documents luôn public (không cần Premium)
      allow read: if request.auth != null;
      allow create: if request.auth != null && 
                       request.resource.data.uploaderUid == request.auth.uid;
      allow update, delete: if request.auth != null && 
                               resource.data.uploaderUid == request.auth.uid;
    }
    
    // ============================================
    // MATCHES COLLECTION - PREMIUM REQUIRED
    // ============================================
    match /matches/{matchId} {
      // Chỉ cho phép user có Premium truy cập
      allow read: if request.auth != null && 
                     hasPremiumAccess(request.auth.uid) &&
                     (resource.data.user1Uid == request.auth.uid || 
                      resource.data.user2Uid == request.auth.uid);
      
      allow create: if request.auth != null && 
                       hasPremiumAccess(request.auth.uid) &&
                       (request.resource.data.user1Uid == request.auth.uid || 
                        request.resource.data.user2Uid == request.auth.uid);
      
      allow update, delete: if request.auth != null && 
                               hasPremiumAccess(request.auth.uid) &&
                               (resource.data.user1Uid == request.auth.uid || 
                                resource.data.user2Uid == request.auth.uid);
    }
    
    // ============================================
    // DATING_PROFILES COLLECTION - PREMIUM REQUIRED
    // ============================================
    match /dating_profiles/{profileId} {
      // Chỉ cho phép user có Premium truy cập
      allow read: if request.auth != null && 
                     hasPremiumAccess(request.auth.uid);
      
      allow write: if request.auth != null && 
                      hasPremiumAccess(request.auth.uid) &&
                      request.auth.uid == profileId;
    }
    
    // ============================================
    // MESSAGES COLLECTION - PREMIUM REQUIRED
    // ============================================
    match /messages/{messageId} {
      // Chỉ cho phép user có Premium truy cập tin nhắn
      allow read: if request.auth != null && 
                     hasPremiumAccess(request.auth.uid) &&
                     (resource.data.senderUid == request.auth.uid || 
                      resource.data.receiverUid == request.auth.uid);
      
      allow create: if request.auth != null && 
                       hasPremiumAccess(request.auth.uid) &&
                       request.resource.data.senderUid == request.auth.uid;
      
      allow update, delete: if request.auth != null && 
                               hasPremiumAccess(request.auth.uid) &&
                               resource.data.senderUid == request.auth.uid;
    }
    
    // ============================================
    // CONVERSATIONS COLLECTION - PREMIUM REQUIRED
    // ============================================
    match /conversations/{conversationId} {
      // Chỉ cho phép user có Premium truy cập
      allow read: if request.auth != null && 
                     hasPremiumAccess(request.auth.uid) &&
                     request.auth.uid in resource.data.participants;
      
      allow write: if request.auth != null && 
                      hasPremiumAccess(request.auth.uid) &&
                      request.auth.uid in request.resource.data.participants;
    }
    
    // ============================================
    // DEFAULT DENY ALL
    // ============================================
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
